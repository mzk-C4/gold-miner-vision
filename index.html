class IGestureRecognizer {
public:
    virtual bool start() = 0;           // 启动识别器
    virtual void stop() = 0;            // 停止识别器
    virtual void pushFrame(...) = 0;    // 推送一帧图像
    virtual void setCallback(...) = 0; // 设置回调
};

// 本地实现 — OpenCV 颜色阈值识别
class LocalGestureRecognizer : public IGestureRecognizer { ... };

// 云端实现 — HTTP API 调用
class CloudGestureRecognizer : public IGestureRecognizer { ... };

// 融合器 — 取置信度最高的结果
class GestureFusion : public IGestureRecognizer { ... };
